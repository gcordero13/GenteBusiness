# Horarios (Work Schedules) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let every contact have an effective work schedule (a company-wide default, optionally overridden per person), surfaced on their profile as "Horario: Lunes a viernes, 8:00 am – 5:00 pm", editable by a new "Horarios" admin module.

**Architecture:** Three additive Supabase migrations (a new `company_schedule` singleton table + RLS, a new `schedules` module registration, four nullable override columns on `contacts` + an updated `enforce_contacts_update_permissions()` trigger). A new pure-function library `src/lib/schedules.ts` computes the "effective" schedule (override-or-default) and formats it for display — this is the only place that fallback logic lives. A new admin page ("Horarios") edits the singleton row, following the exact pattern of the existing `/settings` page. `ContactForm` gains four optional override fields. Three existing profile-display components each gain one derived "Horario:" line.

**Tech Stack:** Next.js 16 (App Router, Server Actions), Supabase (Postgres + RLS), Vitest (`bun run test` in this worktree — `node.exe` is currently missing on this machine, so use `bun run <script>` instead of `npm run <script>` everywhere, and `bunx tsc --noEmit` for type-checking).

- [ ] **Step 1: Write the migration**

This adds the four override columns and re-defines `enforce_contacts_update_permissions()` to admin-gate them (same pattern `hire_date` and `hikvision_employee_no` already follow — copy the current function body from `supabase/migrations/20260810130000_contacts_hikvision_employee_no_unique.sql`'s equivalent, and add the four new columns to the `can_edit`-gated tuple):

```sql
alter table public.contacts add column schedule_start_time time;
alter table public.contacts add column schedule_end_time time;
alter table public.contacts add column schedule_workdays smallint[];
alter table public.contacts add column schedule_tolerance_minutes integer;

create or replace function public.enforce_contacts_update_permissions()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
  flags record;
  is_self boolean;
begin
  select * into flags from public.get_my_module_permissions('contacts');

  select exists (
    select 1 from public.app_users au
    where au.id = auth.uid() and au.email = old.email
  ) into is_self;

  if flags is null and not is_self then
    raise exception 'not authorized';
  end if;

  if new.status is distinct from old.status and not coalesce(flags.can_deactivate, false) then
    raise exception 'not authorized to change contact status';
  end if;

  if (
    new.position, new.fleet_phone, new.extension, new.has_whatsapp
  ) is distinct from (
    old.position, old.fleet_phone, old.extension, old.has_whatsapp
  ) and not (is_self or coalesce(flags.can_edit, false)) then
    raise exception 'not authorized to edit contact fields';
  end if;

  if (
    new.first_name, new.last_name, new.email, new.department_id, new.company_id,
    new.photo_url, new.reports_to_id, new.birth_date, new.hire_date, new.hikvision_employee_no,
    new.schedule_start_time, new.schedule_end_time, new.schedule_workdays, new.schedule_tolerance_minutes
  ) is distinct from (
    old.first_name, old.last_name, old.email, old.department_id, old.company_id,
    old.photo_url, old.reports_to_id, old.birth_date, old.hire_date, old.hikvision_employee_no,
    old.schedule_start_time, old.schedule_end_time, old.schedule_workdays, old.schedule_tolerance_minutes
  ) and not coalesce(flags.can_edit, false) then
    raise exception 'not authorized to edit contact fields';
  end if;

  new.updated_at = now();
  return new;
end;
$$;
```

```typescript
export interface CompanySchedule {
  start_time: string;
  end_time: string;
  workdays: number[];
  tolerance_minutes: number;
}

export interface ScheduleOverrideFields {
  schedule_start_time: string | null;
  schedule_end_time: string | null;
  schedule_workdays: number[] | null;
  schedule_tolerance_minutes: number | null;
}

// Only schedule_start_time is checked. The contact form enforces an
// all-or-nothing rule (all four fields set together, or none), so in
// practice this is equivalent to checking all four. If schedule_start_time
// is null - even if a sibling field is unexpectedly set (e.g. edited
// directly in the database) - this falls back entirely to the company
// default rather than mixing the present sibling field with defaults for
// the rest.
export function getEffectiveSchedule(
  override: ScheduleOverrideFields,
  companyDefault: CompanySchedule,
): CompanySchedule {
  if (!override.schedule_start_time) {
    return companyDefault;
  }
  return {
    start_time: override.schedule_start_time,
    end_time: override.schedule_end_time as string,
    workdays: override.schedule_workdays as number[],
    tolerance_minutes: override.schedule_tolerance_minutes as number,
  };
}

export function formatScheduleTime(time: string): string {
  const [hourStr, minuteStr] = time.split(":");
  const hour = Number(hourStr);
  const minute = Number(minuteStr);
  const period = hour >= 12 ? "pm" : "am";
  const displayHour = hour % 12 === 0 ? 12 : hour % 12;
  return `${displayHour}:${String(minute).padStart(2, "0")} ${period}`;
}

const DAY_NAMES_ES = ["domingo", "lunes", "martes", "miércoles", "jueves", "viernes", "sábado"];

function capitalize(text: string): string {
  return text.charAt(0).toUpperCase() + text.slice(1);
}

export function formatWorkdays(workdays: number[]): string {
  const sorted = [...workdays].sort((a, b) => a - b);
  const isContiguousRange =
    sorted.length > 1 && sorted.every((day, i) => i === 0 || day === sorted[i - 1] + 1);

  if (isContiguousRange) {
    return capitalize(`${DAY_NAMES_ES[sorted[0]]} a ${DAY_NAMES_ES[sorted[sorted.length - 1]]}`);
  }

  return capitalize(sorted.map((day) => DAY_NAMES_ES[day]).join(", "));
}

export function formatScheduleLine(schedule: CompanySchedule): string {
  return `${formatWorkdays(schedule.workdays)}, ${formatScheduleTime(schedule.start_time)} – ${formatScheduleTime(schedule.end_time)}`;
}
```

```typescript
"use server";

import { revalidatePath } from "next/cache";
import { createClient } from "@/lib/supabase/server";

export interface SaveCompanyScheduleInput {
  start_time: string;
  end_time: string;
  workdays: number[];
  tolerance_minutes: number;
}

interface ActionResult {
  error?: string;
}

export async function saveCompanySchedule(
  input: SaveCompanyScheduleInput,
): Promise<ActionResult> {
  const supabase = await createClient();
  const { error } = await supabase
    .from("company_schedule")
    .update({
      start_time: input.start_time,
      end_time: input.end_time,
      workdays: input.workdays,
      tolerance_minutes: input.tolerance_minutes,
    })
    .eq("id", true);

  if (error) return { error: error.message };

  revalidatePath("/schedules");
  return {};
}
```

```typescript
"use client";

import { useState, useTransition } from "react";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { saveCompanySchedule } from "./actions";

const DAY_LABELS = ["Dom", "Lun", "Mar", "Mié", "Jue", "Vie", "Sáb"];

export interface ScheduleFormInitial {
  start_time: string;
  end_time: string;
  workdays: number[];
  tolerance_minutes: number;
}

export function ScheduleForm({ initial }: { initial: ScheduleFormInitial }) {
  const [startTime, setStartTime] = useState(initial.start_time.slice(0, 5));
  const [endTime, setEndTime] = useState(initial.end_time.slice(0, 5));
  const [workdays, setWorkdays] = useState<number[]>(initial.workdays);
  const [toleranceMinutes, setToleranceMinutes] = useState(String(initial.tolerance_minutes));
  const [error, setError] = useState<string | null>(null);
  const [success, setSuccess] = useState(false);
  const [isPending, startTransition] = useTransition();

  function toggleDay(day: number) {
    setWorkdays((prev) =>
      prev.includes(day) ? prev.filter((d) => d !== day) : [...prev, day].sort((a, b) => a - b),
    );
  }

  function submit() {
    startTransition(async () => {
      const result = await saveCompanySchedule({
        start_time: startTime,
        end_time: endTime,
        workdays,
        tolerance_minutes: Number(toleranceMinutes) || 0,
      });
      setError(result.error ?? null);
      setSuccess(!result.error);
    });
  }

  const canSubmit = startTime && endTime && workdays.length > 0;

  return (
    <div className="max-w-md space-y-4 rounded-lg border p-4">
      {error && <p className="text-sm text-red-600">{error}</p>}
      {success && <p className="text-sm text-emerald-600">Horario guardado.</p>}
      <div className="grid grid-cols-2 gap-2">
        <div className="space-y-1">
          <Label>Hora de entrada</Label>
          <Input type="time" value={startTime} onChange={(e) => setStartTime(e.target.value)} />
        </div>
        <div className="space-y-1">
          <Label>Hora de salida</Label>
          <Input type="time" value={endTime} onChange={(e) => setEndTime(e.target.value)} />
        </div>
      </div>
      <div className="space-y-1">
        <Label>Días laborables</Label>
        <div className="flex gap-3">
          {DAY_LABELS.map((label, day) => (
            <label key={day} className="flex flex-col items-center gap-1 text-xs">
              <input
                type="checkbox"
                checked={workdays.includes(day)}
                onChange={() => toggleDay(day)}
              />
              {label}
            </label>
          ))}
        </div>
      </div>
      <div className="space-y-1">
        <Label>Minutos de tolerancia</Label>
        <Input
          type="number"
          min={0}
          value={toleranceMinutes}
          onChange={(e) => setToleranceMinutes(e.target.value)}
        />
      </div>
      <Button onClick={submit} disabled={isPending || !canSubmit}>
        Guardar
      </Button>
    </div>
  );
}
```

```typescript
import { afterEach, describe, expect, it } from "vitest";
import { createAdminClient } from "@/lib/supabase/admin";
import { createTestUser, deleteTestUser, type TestUser } from "./supabaseTestHelpers";

const DEFAULT_SCHEDULE = {
  start_time: "08:00:00",
  end_time: "17:00:00",
  workdays: [1, 2, 3, 4, 5],
  tolerance_minutes: 0,
};

describe("company_schedule RLS", () => {
  let viewer: TestUser | undefined;
  let admin: TestUser | undefined;

  afterEach(async () => {
    if (viewer) await deleteTestUser(viewer.id);
    if (admin) await deleteTestUser(admin.id);
    viewer = undefined;
    admin = undefined;

    // company_schedule is a permanent singleton row, not a row this suite
    // creates and tears down - reset it after every test so a mutation in
    // one test can't leak into another.
    const adminClient = createAdminClient();
    await adminClient.from("company_schedule").update(DEFAULT_SCHEDULE).eq("id", true);
  });

  it("lets any authenticated user read company_schedule", async () => {
    viewer = await createTestUser("Viewer");

    const { data, error } = await viewer.client
      .from("company_schedule")
      .select("*")
      .eq("id", true)
      .single();

    expect(error).toBeNull();
    expect(data?.start_time).toBe(DEFAULT_SCHEDULE.start_time);
  });

  it("blocks a Viewer (no schedules can_manage) from updating company_schedule", async () => {
    viewer = await createTestUser("Viewer");

    // The update/select policies are symmetric (both require can_manage), so
    // Postgres RLS filters the row out of the update's target set silently -
    // 0 rows affected, no error - rather than raising a permission error.
    // Verify via a follow-up admin read instead of asserting on `error`.
    await viewer.client.from("company_schedule").update({ tolerance_minutes: 5 }).eq("id", true);

    const adminClient = createAdminClient();
    const { data } = await adminClient
      .from("company_schedule")
      .select("tolerance_minutes")
      .eq("id", true)
      .single();
    expect(data?.tolerance_minutes).toBe(DEFAULT_SCHEDULE.tolerance_minutes);
  });

  it("lets a Super Admin update company_schedule", async () => {
    admin = await createTestUser("Super Admin");

    const { data, error } = await admin.client
      .from("company_schedule")
      .update({ tolerance_minutes: 10 })
      .eq("id", true)
      .select()
      .single();

    expect(error).toBeNull();
    expect(data?.tolerance_minutes).toBe(10);
  });
});
```
